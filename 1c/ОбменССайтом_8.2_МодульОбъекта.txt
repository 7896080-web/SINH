// Главная процедура обмена: разобрать все задания из каталога tasks.
Процедура ВыполнитьОбмен() Экспорт
	Пар = ПараметрыОбмена();
	Соединение = ПодключитьсяFTP(Пар);

	ФайлыЗаданий = Соединение.НайтиФайлы(Пар.КаталогTasks, "task_*.txt");
	Для Каждого ФайлЗадания Из ФайлыЗаданий Цикл
		Попытка
			ОбработатьФайлЗадания(Соединение, Пар, ФайлЗадания);
		Исключение
			ЗаписьЖурналаРегистрации("ОбменССайтом", УровеньЖурналаРегистрации.Ошибка, , ,
				"Ошибка обработки " + ФайлЗадания.Имя + ": " + ОписаниеОшибки());
		КонецПопытки;
	КонецЦикла;
КонецПроцедуры

// Обработать один файл-задание: выполнить команды, записать результаты и/или
// выгрузку остатков, перенести файл в архив.
Процедура ОбработатьФайлЗадания(Соединение, Пар, ФайлЗадания)
	ЛокальныйФайл = ПолучитьИмяВременногоФайла("txt");
	Соединение.Получить(ФайлЗадания.ПолноеИмя, ЛокальныйФайл);
	Содержимое = ПрочитатьТекстФайла(ЛокальныйФайл);

	Метка = Формат(ТекущаяДата(), "ДФ=ггггММддЧЧммсс");
	СтрокиРезультата = Новый Массив;
	НужнаВыгрузкаОстатков = Ложь;

	Для Каждого Строка Из РазложитьСтроку(Содержимое, Символы.ПС) Цикл
		Строка = СокрЛП(СтрЗаменить(Строка, Символы.ВК, ""));
		Если ПустаяСтрока(Строка) Тогда
			Продолжить;
		КонецЕсли;
		Поля = РазложитьСтроку(Строка, "|");
		Команда = Поля[0];

		Если Команда = "EXPORT_STOCK_ON_HAND" Тогда
			НужнаВыгрузкаОстатков = Истина;

		ИначеЕсли Команда = "CREATE_MOVEMENT" ИЛИ Команда = "CONFIRM_MOVEMENT" Тогда
			// Поля: 0=команда 1=баркод 2=склад_откуда 3=склад_куда 4=кол-во 5=order_id 6=площадка
			ИдЗаказа = Поля[5];
			Попытка
				НомерДок = ВыполнитьПеремещение(Команда, Поля[1], Поля[2], Поля[3], Число(Поля[4]), ИдЗаказа, Поля[6]);
				СтрокиРезультата.Добавить(ИдЗаказа + "|OK|" + НомерДок);
			Исключение
				СтрокиРезультата.Добавить(ИдЗаказа + "|ERROR|" + ОписаниеОшибки());
			КонецПопытки;

		ИначеЕсли Команда = "CANCEL_MOVEMENT" Тогда
			// Поля: 0=команда 1=order_id 2=площадка
			ИдЗаказа = Поля[1];
			Попытка
				НомерДок = ОтменитьПеремещенияЗаказа(ИдЗаказа);
				СтрокиРезультата.Добавить(ИдЗаказа + "|OK|" + НомерДок);
			Исключение
				СтрокиРезультата.Добавить(ИдЗаказа + "|ERROR|" + ОписаниеОшибки());
			КонецПопытки;
		КонецЕсли;
	КонецЦикла;

	Если СтрокиРезультата.Количество() > 0 Тогда
		ОпубликоватьФайл(Соединение, Пар.КаталогResults, "result_" + Метка + ".txt", СоединитьМассив(СтрокиРезультата, Символы.ПС));
	КонецЕсли;

	Если НужнаВыгрузкаОстатков Тогда
		ОпубликоватьФайл(Соединение, Пар.КаталогResults, "stock_" + Метка + ".txt", ВыгрузитьОстаткиЦС(Пар));
	КонецЕсли;

	// Перенести файл задания в архив: 8.2 не умеет Переместить по FTP —
	// делаем как загрузка в архив + удаление источника.
	ПереместитьНаFTP(Соединение, ФайлЗадания.ПолноеИмя, СоединитьПуть(Пар.КаталогArchive, ФайлЗадания.Имя), ЛокальныйФайл);
	УдалитьФайлы(ЛокальныйФайл);
КонецПроцедуры

// ---- Бизнес-операции. НАСТРОИТЬ под конфигурацию «Магазин одежды и обуви». ----

// НАСТРОИТЬ: создание/проведение документа перемещения (ЦС->Ожидает или
// Ожидает->Склад площадки — разницы в логике нет). order_id сохранить в
// документе (реквизит/комментарий) — по нему работает отмена.
Функция ВыполнитьПеремещение(Команда, Баркод, ИмяСкладаОткуда, ИмяСкладаКуда, Количество, ИдЗаказа, Площадка)
	СведенияОТоваре = НайтиНоменклатуруПоШтрихкоду(Баркод);
	Если СведенияОТоваре = Неопределено Тогда
		ВызватьИсключение "Не найдена номенклатура по штрихкоду " + Баркод;
	КонецЕсли;

	СкладОткуда = НайтиСклад(ИмяСкладаОткуда);
	СкладКуда   = НайтиСклад(ИмяСкладаКуда);
	Документ = Документы.ПеремещениеТоваров.СоздатьДокумент();
	Документ.Дата = ТекущаяДата();
	Документ.СкладОтправитель = СкладОткуда;
	Документ.СкладПолучатель  = СкладКуда;
	Документ.МагазинОтправитель = СкладОткуда.Магазин;   // Магазин одежды 1.0: склад подчинён магазину
	Документ.МагазинПолучатель  = СкладКуда.Магазин;
	Документ.Комментарий = "sync order_id=" + ИдЗаказа + " " + Площадка;
	Строка = Документ.Товары.Добавить();
	Строка.Номенклатура = СведенияОТоваре.Номенклатура;
	Строка.Характеристика = СведенияОТоваре.Характеристика;
	Строка.Количество = Количество;
	Документ.Записать(РежимЗаписиДокумента.Проведение);

	Возврат Строка(Документ.Номер);
КонецФункции

// НАСТРОИТЬ: отмена неподтверждённого заказа — реверс движений на ЦС.
Функция ОтменитьПеремещенияЗаказа(ИдЗаказа)
	Запрос = Новый Запрос;
	Запрос.Текст =
		"ВЫБРАТЬ Ссылка ИЗ Документ.ПеремещениеТоваров
		|ГДЕ Комментарий ПОДОБНО &Шаблон И Проведен = ИСТИНА";   // НАСТРОИТЬ
	Запрос.УстановитьПараметр("Шаблон", "%order_id=" + ИдЗаказа + " %");
	Выборка = Запрос.Выполнить().Выбрать();

	НомераОбратных = Новый Массив;
	Пока Выборка.Следующий() Цикл
		Исходный = Выборка.Ссылка.ПолучитьОбъект();
		СкладЦелевой = НайтиСклад(ПараметрыОбмена().СкладЦС);
		Обратный = Документы.ПеремещениеТоваров.СоздатьДокумент();
		Обратный.Дата = ТекущаяДата();
		Обратный.СкладОтправитель = Исходный.СкладПолучатель;
		Обратный.СкладПолучатель  = СкладЦелевой;
		Обратный.МагазинОтправитель = Исходный.МагазинПолучатель;
		Обратный.МагазинПолучатель  = СкладЦелевой.Магазин;
		Обратный.Комментарий = "sync REVERSE " + Исходный.Комментарий;
		Для Каждого СтрокаТовара Из Исходный.Товары Цикл
			НоваяСтрока = Обратный.Товары.Добавить();
			ЗаполнитьЗначенияСвойств(НоваяСтрока, СтрокаТовара);
		КонецЦикла;
		Обратный.Записать(РежимЗаписиДокумента.Проведение);
		НомераОбратных.Добавить(Строка(Обратный.Номер));
	КонецЦикла;

	Если НомераОбратных.Количество() = 0 Тогда
		Возврат "нет документов для отмены";
	КонецЕсли;
	Возврат СоединитьМассив(НомераОбратных, ",");
КонецФункции

// НАСТРОИТЬ: выгрузка остатков склада ЦС.
// Строка: uid_1c|артикул|наименование|количество|баркод1,баркод2
Функция ВыгрузитьОстаткиЦС(Пар)
	Запрос = Новый Запрос;
	Запрос.Текст =
		"ВЫБРАТЬ
		|	Остатки.Номенклатура КАК Номенклатура,
		|	Остатки.ХарактеристикаНоменклатуры КАК Характеристика,
		|	Остатки.КоличествоОстаток КАК Количество
		|ИЗ
		|	РегистрНакопления.ТоварыНаСкладах.Остатки(, Склад = &Склад) КАК Остатки"; // Розница: поле ХарактеристикаНоменклатуры; подтвердить измерение Склад
	Запрос.УстановитьПараметр("Склад", НайтиСклад(Пар.СкладЦС));
	Выборка = Запрос.Выполнить().Выбрать();

	Строки = Новый Массив;
	Пока Выборка.Следующий() Цикл
		Уид = ИдентификаторSKU(Выборка.Номенклатура, Выборка.Характеристика); // НАСТРОИТЬ
		Артикул = "" + Выборка.Номенклатура.Артикул;                          // НАСТРОИТЬ имя реквизита
		Наименование = "" + Выборка.Номенклатура.Наименование;
		Кол = Формат(Выборка.Количество, "ЧГ=0; ЧН=-; ЧДЦ=0");
		Баркоды = БаркодыSKU(Выборка.Номенклатура, Выборка.Характеристика);   // НАСТРОИТЬ "bc1,bc2"
		Если ПустаяСтрока(Баркоды) Тогда
			Продолжить;
		КонецЕсли;
		Строки.Добавить(Уид + "|" + Артикул + "|" + Наименование + "|" + Кол + "|" + Баркоды);
	КонецЦикла;

	Возврат СоединитьМассив(Строки, Символы.ПС);
КонецФункции

// НАСТРОИТЬ: поиск номенклатуры (и характеристики) по штрихкоду.
Функция НайтиНоменклатуруПоШтрихкоду(Баркод)
	Запрос = Новый Запрос;
	Запрос.Текст =
		"ВЫБРАТЬ ПЕРВЫЕ 1
		|	Владелец КАК Номенклатура,
		|	ХарактеристикаНоменклатуры КАК Характеристика
		|ИЗ РегистрСведений.Штрихкоды
		|ГДЕ Штрихкод = &Штрихкод";                                  // Розница: поля Владелец / ХарактеристикаНоменклатуры
	Запрос.УстановитьПараметр("Штрихкод", Баркод);
	Выборка = Запрос.Выполнить().Выбрать();
	Если НЕ Выборка.Следующий() Тогда
		Возврат Неопределено;
	КонецЕсли;
	Рез = Новый Структура("Номенклатура, Характеристика");
	Рез.Номенклатура = Выборка.Номенклатура;
	Рез.Характеристика = Выборка.Характеристика;
	Возврат Рез;
КонецФункции

// НАСТРОИТЬ: идентификатор размер-цвет SKU (Product.uid_1c).
Функция ИдентификаторSKU(Номенклатура, Характеристика)
	Если ЗначениеЗаполнено(Характеристика) Тогда
		Возврат Строка(Характеристика.УникальныйИдентификатор());
	КонецЕсли;
	Возврат Строка(Номенклатура.УникальныйИдентификатор());
КонецФункции

// НАСТРОИТЬ: все баркоды размер-цвета через запятую.
Функция БаркодыSKU(Номенклатура, Характеристика)
	Запрос = Новый Запрос;
	Запрос.Текст =
		"ВЫБРАТЬ Штрихкод ИЗ РегистрСведений.Штрихкоды
		|ГДЕ Владелец = &Номенклатура И ХарактеристикаНоменклатуры = &Характеристика"; // Розница: поля Владелец / ХарактеристикаНоменклатуры
	Запрос.УстановитьПараметр("Номенклатура", Номенклатура);
	Запрос.УстановитьПараметр("Характеристика", Характеристика);
	Выборка = Запрос.Выполнить().Выбрать();
	Список = Новый Массив;
	Пока Выборка.Следующий() Цикл
		Если НЕ ПустаяСтрока(Выборка.Штрихкод) Тогда
			Список.Добавить(СокрЛП(Выборка.Штрихкод));
		КонецЕсли;
	КонецЦикла;
	Возврат СоединитьМассив(Список, ",");
КонецФункции

// Поиск склада по наименованию. НАСТРОИТЬ, если склады — не справочник Склады.
Функция НайтиСклад(Имя)
	Склад = Справочники.Склады.НайтиПоНаименованию(Имя, Истина);   // НАСТРОИТЬ
	Если Склад = Неопределено ИЛИ Склад.Пустая() Тогда
		ВызватьИсключение "Не найден склад «" + Имя + "» — заведите его в 1С с точным именем.";
	КонецЕсли;
	Возврат Склад;
КонецФункции

// Публикация файла на FTP (8.2, без атомарного rename): пишем сразу в итог.
Процедура ОпубликоватьФайл(Соединение, КаталогНаСервере, ИмяФайла, Текст)
	Локальный = ПолучитьИмяВременногоФайла("txt");
	ЗаписатьТекстФайла(Локальный, Текст);
	Соединение.Записать(Локальный, СоединитьПуть(КаталогНаСервере, ИмяФайла));
	УдалитьФайлы(Локальный);
КонецПроцедуры

// "Перемещение" файла на FTP в 8.2: файл уже скачан локально (ЛокальнаяКопия);
// заливаем его в приёмник и удаляем источник.
Процедура ПереместитьНаFTP(Соединение, ПолноеИмяИсточника, ПолноеИмяПриёмника, ЛокальнаяКопия)
	Соединение.Записать(ЛокальнаяКопия, ПолноеИмяПриёмника);
	Соединение.Удалить(ПолноеИмяИсточника);
КонецПроцедуры

Функция ПрочитатьТекстФайла(Путь)
	Чтение = Новый ЧтениеТекста(Путь, КодировкаТекста.UTF8);
	Содержимое = Чтение.Прочитать();
	Чтение.Закрыть();
	Если Содержимое = Неопределено Тогда
		Возврат "";
	КонецЕсли;
	Возврат Содержимое;
КонецФункции

Процедура ЗаписатьТекстФайла(Путь, Текст)
	Запись = Новый ЗаписьТекста(Путь, КодировкаТекста.UTF8);
	Запись.Записать(Текст);
	Запись.Закрыть();
КонецПроцедуры

// Разбить строку по разделителю (в 8.2 нет СтрРазделить).
Функция РазложитьСтроку(Строка, Разделитель)
	Результат = Новый Массив;
	Остаток = Строка;
	Поз = Найти(Остаток, Разделитель);
	Пока Поз > 0 Цикл
		Результат.Добавить(Лев(Остаток, Поз - 1));
		Остаток = Сред(Остаток, Поз + СтрДлина(Разделитель));
		Поз = Найти(Остаток, Разделитель);
	КонецЦикла;
	Результат.Добавить(Остаток);
	Возврат Результат;
КонецФункции

// Соединить массив строк разделителем (в 8.2 нет СтрСоединить).
Функция СоединитьМассив(Массив, Разделитель)
	Результат = "";
	Первый = Истина;
	Для Каждого Элемент Из Массив Цикл
		Если Первый Тогда
			Результат = "" + Элемент;
			Первый = Ложь;
		Иначе
			Результат = Результат + Разделитель + Элемент;
		КонецЕсли;
	КонецЦикла;
	Возврат Результат;
