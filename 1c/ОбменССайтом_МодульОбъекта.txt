// Внешняя обработка "ОбменССайтом" — обмен с приложением синхронизации остатков
// через ЛОКАЛЬНУЮ ПАПКУ (без FTP; 1С, приложение и папки — на одной машине).
// Платформа 1С:Предприятие 8.3 (проверено на 8.3.9.2233).
// Конфигурация «Магазин одежды и обуви» 1.0.16.x. Протокол — см. 1c/README.md.
// Кодировка файлов UTF-8, разделитель полей "|".
//
// Каталоги обмена (локальные): C:\sync\tasks | results | archive.
// Публикация файлов атомарная: пишем "имя.part", затем переименовываем в "имя".
//
// Имена метаданных под базу: РегистрСведений.Штрихкоды (Владелец,
// ХарактеристикаНоменклатуры), РегистрНакопления.ТоварыНаСкладах (Склад,
// Номенклатура, ХарактеристикаНоменклатуры), Документы.ПеремещениеТоваров.
//
// Точка входа — ВыполнитьОбмен(). Кнопкой формы и/или по расписанию.

// Параметры обмена. Пути локальные — при необходимости поправьте.
Функция ПараметрыОбмена()
	Пар = Новый Структура;
	Пар.Вставить("КаталогTasks", "C:\sync\tasks");
	Пар.Вставить("КаталогResults", "C:\sync\results");
	Пар.Вставить("КаталогArchive", "C:\sync\archive");
	Пар.Вставить("СкладЦС", "ЦС Склад");         // центральный склад-источник (магазин ЦС_МСК)
	Возврат Пар;
КонецФункции

// Главная процедура обмена: разобрать все задания из каталога tasks.
Процедура ВыполнитьОбмен() Экспорт
	Пар = ПараметрыОбмена();

	ФайлыЗаданий = НайтиФайлы(Пар.КаталогTasks, "task_*.txt");
	Для Каждого ФайлЗадания Из ФайлыЗаданий Цикл
		Если НЕ ФайлЗадания.ЭтоФайл() Тогда
			Продолжить;
		КонецЕсли;
		Попытка
			ОбработатьФайлЗадания(Пар, ФайлЗадания);
		Исключение
			ЗаписьЖурналаРегистрации("ОбменССайтом", УровеньЖурналаРегистрации.Ошибка, , ,
				"Ошибка обработки " + ФайлЗадания.Имя + ": " + ОписаниеОшибки());
		КонецПопытки;
	КонецЦикла;

	// Оперативная отдача изменений остатка ЦС — СВОЯ Попытка и последним делом.
	// Это добавка к обмену, а не его часть: что бы здесь ни случилось, задания
	// уже разобраны и ответы отданы. Сломать существующий обмен она не может.
	Попытка
		ВыгрузитьИзменениеОстатковЦС(Пар);
	Исключение
		ЗаписьЖурналаРегистрации("ОбменССайтом", УровеньЖурналаРегистрации.Ошибка, , ,
			"Выгрузка изменения остатков ЦС: " + ОписаниеОшибки());
	КонецПопытки;
КонецПроцедуры

// Обработать один файл-задание: выполнить команды, записать результаты и/или
// выгрузку остатков, перенести файл в архив.
Процедура ОбработатьФайлЗадания(Пар, ФайлЗадания)
	Содержимое = ПрочитатьТекстФайла(ФайлЗадания.ПолноеИмя);

	// Метка файлов-ответов берётся из ИМЕНИ ЗАДАНИЯ, а не из текущего времени.
	// ТекущаяДата() имеет точность до секунды, а ВыполнитьОбмен разбирает все
	// накопившиеся задания одним циклом — два файла легко попадают в одну
	// секунду и получают ОДИНАКОВОЕ имя ответа. ОпубликоватьФайл существующий
	// файл удаляет, поэтому второй ответ молча затирал первый, и задания из
	// затёртого оставались непроведёнными навсегда (19.09.2026: так пропали два
	// файла из пятидесяти трёх — 12 перемещений). Имена task_*.txt приложение
	// гарантирует уникальными, значит уникальны и ответы.
	Метка = СтрЗаменить(СтрЗаменить(ФайлЗадания.Имя, "task_", ""), ".txt", "");
	СтрокиРезультата = Новый Массив;
	НужнаВыгрузкаОстатков = Ложь;
	НуженСправочникБаркодов = Ложь;
	ДатыОстатков = Новый Массив;      // даты срезов из команд EXPORT_STOCK_ON_DATE

	Для Каждого СтрокаФайла Из СтрРазделить(Содержимое, Символы.ПС) Цикл
		СтрокаФайла = СокрЛП(СтрЗаменить(СтрокаФайла, Символы.ВК, ""));
		Если ПустаяСтрока(СтрокаФайла) Тогда
			Продолжить;
		КонецЕсли;
		Поля = СтрРазделить(СтрокаФайла, "|");
		Команда = Поля[0];

		Если Команда = "EXPORT_STOCK_ON_HAND" Тогда
			НужнаВыгрузкаОстатков = Истина;

		ИначеЕсли Команда = "EXPORT_BARCODES" Тогда
			НуженСправочникБаркодов = Истина;

		ИначеЕсли Команда = "EXPORT_STOCK_ON_DATE" Тогда
			// Поля: 0=команда 1=дата среза ГГГГММДД. Справка «сколько лежало на
			// ЦС такого-то числа» — читаем регистр остатков, ничего не меняем.
			Если Поля.Количество() > 1 И СтрДлина(СокрЛП(Поля[1])) = 8 Тогда
				ДатыОстатков.Добавить(СокрЛП(Поля[1]));
			КонецЕсли;

		ИначеЕсли Команда = "CREATE_MOVEMENT" ИЛИ Команда = "CONFIRM_MOVEMENT"
			ИЛИ Команда = "RETURN_TO_STOCK" Тогда
			// Поля: 0=команда 1=баркод 2=склад_откуда 3=склад_куда 4=кол-во 5=order_id 6=площадка 7=дата(ГГГГММДД, необяз.)
			// Проверка длины ОБЯЗАТЕЛЬНА и стоит ДО Попытка: обращение Поля[5] к
			// короткой строке роняет разбор файла целиком, а падение здесь — это
			// не только потерянный ответ. Файл задания не архивируется (архив
			// ниже по процедуре), следующий запуск разбирает его заново, и по
			// CANCEL_MOVEMENT это создаёт ЕЩЁ ОДИН обратный документ каждую
			// минуту: отмена идемпотентной не сделана и сделана быть не может.
			Если Поля.Количество() < 7 Тогда
				ЗаписьЖурналаРегистрации("ОбменССайтом", УровеньЖурналаРегистрации.Ошибка, , ,
					"Строка " + Команда + ": полей " + Поля.Количество() + ", нужно 7 — строка пропущена");
				Продолжить;
			КонецЕсли;
			ИдЗаказа = Поля[5];
			ДатаДок = ?(Поля.Количество() > 7, Поля[7], "");
			Попытка
				НомерДок = ВыполнитьПеремещение(Команда, Поля[1], Поля[2], Поля[3], Число(Поля[4]), ИдЗаказа, Поля[6], ДатаДок);
				СтрокиРезультата.Добавить(ИдЗаказа + "|OK|" + НомерДок + "|" + Команда);
			Исключение
				СтрокиРезультата.Добавить(ИдЗаказа + "|ERROR|" + ОписаниеОшибки() + "|" + Команда);
			КонецПопытки;

		ИначеЕсли Команда = "SCRAP_RETURN" Тогда
			// Поля: 0=команда 1=баркод 2=склад_откуда 3=склад_куда 4=кол-во
			//       5=RET-номер 6=площадка 7=дата(необяз.)
			//       8=хоз. операция 9=ответственный
			//
			// ДВА документа одной транзакцией: возврат «склад площадки → ЦС» и
			// списание с ЦС. Порознь их слать нельзя — между ними остаток ЦС
			// вырастет на единицу, и часовая выгрузка, попав в промежуток,
			// увезёт на площадки вещь, которой уже нет физически.
			Если Поля.Количество() < 7 Тогда
				ЗаписьЖурналаРегистрации("ОбменССайтом", УровеньЖурналаРегистрации.Ошибка, , ,
					"Строка SCRAP_RETURN: полей " + Поля.Количество() + ", нужно 7 — строка пропущена");
				Продолжить;
			КонецЕсли;
			ИдЗаказа = Поля[5];
			ДатаДок = ?(Поля.Количество() > 7, Поля[7], "");
			// Девятое поле — НАИМЕНОВАНИЕ хоз. операции списания («Утилизация
			// Брака», «Утилизация Подмены», …). Оно определяет проводки, поэтому
			// приезжает от сайта, а не выбирается здесь: причину знает он.
			ХозОперация = ?(Поля.Количество() > 8, СокрЛП(Поля[8]), "");
			// Десятое поле — ОТВЕТСТВЕННЫЙ, имя пользователя 1С. Приезжает от
			// сайта по той же причине, что и хоз. операция: человек уходит в
			// отпуск и увольняется, а правка этого модуля требует Конфигуратора.
			Ответственный = ?(Поля.Количество() > 9, СокрЛП(Поля[9]), "");
			Попытка
				НомерДок = УтилизироватьВозврат(Поля[1], Поля[2], Поля[3], Число(Поля[4]),
					ИдЗаказа, Поля[6], ДатаДок, ХозОперация, Ответственный);
				СтрокиРезультата.Добавить(ИдЗаказа + "|OK|" + НомерДок + "|SCRAP_RETURN");
			Исключение
				СтрокиРезультата.Добавить(ИдЗаказа + "|ERROR|" + ОписаниеОшибки() + "|SCRAP_RETURN");
			КонецПопытки;

		ИначеЕсли Команда = "CANCEL_MOVEMENT" Тогда
			// Поля: 0=команда 1=order_id 2=площадка
			Если Поля.Количество() < 2 Тогда
				ЗаписьЖурналаРегистрации("ОбменССайтом", УровеньЖурналаРегистрации.Ошибка, , ,
					"Строка CANCEL_MOVEMENT: полей " + Поля.Количество() + ", нужно 2 — строка пропущена");
				Продолжить;
			КонецЕсли;
			ИдЗаказа = Поля[1];
			Попытка
				НомерДок = ОтменитьПеремещенияЗаказа(ИдЗаказа);
				СтрокиРезультата.Добавить(ИдЗаказа + "|OK|" + НомерДок + "|CANCEL_MOVEMENT");
			Исключение
				СтрокиРезультата.Добавить(ИдЗаказа + "|ERROR|" + ОписаниеОшибки() + "|CANCEL_MOVEMENT");
			КонецПопытки;
		КонецЕсли;
	КонецЦикла;

	Если СтрокиРезультата.Количество() > 0 Тогда
		ОпубликоватьФайл(Пар.КаталогResults, "result_" + Метка + ".txt", СтрСоединить(СтрокиРезультата, Символы.ПС));
	КонецЕсли;

	// Архивируем задание ЗДЕСЬ, а не в конце процедуры. Документы к этому моменту
	// уже созданы и проведены, ответ отдан — дело сделано. Раньше архивирование
	// стояло после выгрузок остатков, а это самые тяжёлые запросы в обработке:
	// исключение на них оставляло файл в tasks, и следующий запуск разбирал его
	// заново, создавая ДУБЛИ документов перемещения. Часовой файл от приложения
	// как раз содержит и EXPORT_STOCK_ON_HAND, и строки перемещений сразу.
	// Не получилась выгрузка остатков — приложение попросит её следующим часом,
	// это не стоит ни одного документа.
	АрхивироватьЗадание(Пар, ФайлЗадания);

	Если НужнаВыгрузкаОстатков Тогда
		ОпубликоватьФайл(Пар.КаталогResults, "stock_" + Метка + ".txt", ВыгрузитьОстаткиЦС(Пар));
	КонецЕсли;

	Если НуженСправочникБаркодов Тогда
		ОпубликоватьФайл(Пар.КаталогResults, "barcodes_" + Метка + ".txt", ВыгрузитьСправочникБаркодов(Пар));
	КонецЕсли;

	// Остатки на заданное число. Префикс "ondate_", а НЕ "stock_", и это принципиально:
	// файлы "stock_" приложение применяет как ТЕКУЩИЙ остаток и рассылает на площадки,
	// так что срез за прошлое число под тем же именем отправил бы в WB/Ozon/Kit цифры
	// той давности, которую оператор запросил всего лишь для справки.
	// Сбой одной даты не должен рушить остальной обмен — поэтому каждая в своей Попытке.
	Для Каждого ДатаСрезаСтрокой Из ДатыОстатков Цикл
		Попытка
			ОпубликоватьФайл(Пар.КаталогResults, "ondate_" + ДатаСрезаСтрокой + "_" + Метка + ".txt",
				ВыгрузитьОстаткиЦСНаДату(Пар, ДатаСрезаСтрокой));
		Исключение
			ЗаписьЖурналаРегистрации("ОбменССайтом", УровеньЖурналаРегистрации.Ошибка, , ,
				"Выгрузка остатков на " + ДатаСрезаСтрокой + ": " + ОписаниеОшибки());
		КонецПопытки;
	КонецЦикла;

КонецПроцедуры

// Перенести задание в архив (локальное перемещение).
Процедура АрхивироватьЗадание(Пар, ФайлЗадания)
	ПутьВАрхиве = СоединитьПуть(Пар.КаталогArchive, ФайлЗадания.Имя);
	УдалитьЕслиЕсть(ПутьВАрхиве);
	ПереместитьФайл(ФайлЗадания.ПолноеИмя, ПутьВАрхиве);
КонецПроцедуры

// ---- Бизнес-операции ----

// Создание/проведение документа перемещения (ЦС->Ожидает или Ожидает->Склад
// площадки — логика одна). order_id сохраняется в комментарии — по нему работает отмена.
// ДатаДок (ГГГГММДД, необяз.) — старт задним числом; пусто → текущая дата.
Функция ВыполнитьПеремещение(Команда, Баркод, ИмяСкладаОткуда, ИмяСкладаКуда, Количество, ИдЗаказа, Площадка, ДатаДок = "")
	// Идемпотентность: документ по этому заказу И на этот склад-получатель уже
	// проведён — второй раз не создаём, возвращаем его номер. Повтор задания это
	// не ошибка приложения, а способ УЗНАТЬ, дошёл ли документ, когда ответ на
	// первую попытку потерялся. Без этой проверки повторная отправка заводила бы
	// ВТОРОЙ документ перемещения на тот же заказ.
	//
	// Склад-получатель в условии обязателен: у CREATE_MOVEMENT (ЦС → склад
	// площадки) и CONFIRM_MOVEMENT (склад «в пути» → склад продаж) один и тот же
	// номер заказа в комментарии, и проверка без склада запретила бы законное
	// второе перемещение. Реверсы исключаем отдельно — их комментарий начинается
	// на "sync REVERSE" и содержит тот же order_id.
	СуществующийНомер = НайтиПроведённоеПеремещение(ИдЗаказа, ИмяСкладаКуда);
	Если ЗначениеЗаполнено(СуществующийНомер) Тогда
		Возврат СуществующийНомер;
	КонецЕсли;

	СведенияОТоваре = НайтиНоменклатуруПоШтрихкоду(Баркод);
	Если СведенияОТоваре = Неопределено Тогда
		ВызватьИсключение "Не найдена номенклатура по штрихкоду " + Баркод;
	КонецЕсли;

	СкладОткуда = НайтиСклад(ИмяСкладаОткуда);
	СкладКуда   = НайтиСклад(ИмяСкладаКуда);
	Документ = Документы.ПеремещениеТоваров.СоздатьДокумент();
	// Дата документа: если пришла дата ГГГГММДД — ставим её (старт задним числом),
	// иначе текущую. Запись неоперативная (РежимЗаписиДокумента.Проведение) —
	// проведение задним числом допускается (в пределах даты запрета изменения).
	Если СтрДлина(СокрЛП(ДатаДок)) = 8 Тогда
		Документ.Дата = Дата(Число(Лев(ДатаДок, 4)), Число(Сред(ДатаДок, 5, 2)), Число(Прав(ДатаДок, 2)));
	Иначе
		Документ.Дата = ТекущаяДата();
	КонецЕсли;
	Документ.СкладОтправитель = СкладОткуда;
	Документ.СкладПолучатель  = СкладКуда;
	Документ.МагазинОтправитель = СкладОткуда.Магазин;   // склад подчинён магазину
	Документ.МагазинПолучатель  = СкладКуда.Магазин;
	Документ.Комментарий = "sync order_id=" + ИдЗаказа + " " + Площадка;
	СтрокаТовара = Документ.Товары.Добавить();
	СтрокаТовара.Номенклатура = СведенияОТоваре.Номенклатура;
	СтрокаТовара.ХарактеристикаНоменклатуры = СведенияОТоваре.Характеристика;
	СтрокаТовара.Количество = Количество;
	СтрокаТовара.ЕдиницаИзмерения =
	СведенияОТоваре.Номенклатура.ЕдиницаХраненияОстатков;
    СтрокаТовара.Коэффициент = 1;
	Документ.Записать(РежимЗаписиДокумента.Проведение);

	Возврат Строка(Документ.Номер);
КонецФункции

// Утилизация возврата: ДВА документа одной транзакцией.
//
// Зачем два. Единица уехала на склад площадки перемещением при приёме заказа —
// в 1С она ЧИСЛИТСЯ ТАМ. Покупатель вернул её, склад выбросил, и если ничего не
// проводить, она висит на складе площадки вечно: учёт расходится со складом
// накопительно, по штуке на каждую утилизацию. Поэтому сначала возвращаем её на
// ЦС, потом списываем оттуда.
//
// Зачем ОДНОЙ транзакцией. Между документами остаток ЦС на единицу больше
// правды. Сайт спрашивает остатки ЦС раз в час и минутной дельтой; попади
// запрос в этот промежуток — вещь уедет на площадки как доступная, а её уже
// нет. Промежуточного состояния не должно существовать вовсе.
Функция УтилизироватьВозврат(Баркод, ИмяСкладаОткуда, ИмяСкладаКуда, Количество, ИдЗаказа, Площадка, ДатаДок = "", ХозОперация = "", Ответственный = "")
	// Идемпотентность спрашиваем у СПИСАНИЯ, а не у перемещения: списание идёт
	// вторым, значит его наличие означает, что прошла вся транзакция целиком.
	// Спроси мы перемещение, повтор после сбоя на списании увидел бы «уже
	// сделано» и вернул бы OK, не списав ничего.
	СуществующийНомер = НайтиПроведённоеСписаниеВозврата(ИдЗаказа);
	Если ЗначениеЗаполнено(СуществующийНомер) Тогда
		Возврат СуществующийНомер;
	КонецЕсли;

	НачатьТранзакцию();
	Попытка
		НомерПеремещения = ВыполнитьПеремещение("RETURN_TO_STOCK", Баркод, ИмяСкладаОткуда,
			ИмяСкладаКуда, Количество, ИдЗаказа, Площадка, ДатаДок);
		НомерСписания = СписатьВозврат(Баркод, ИмяСкладаКуда, Количество, ИдЗаказа,
			Площадка, ДатаДок, ХозОперация, Ответственный);
		ЗафиксироватьТранзакцию();
	Исключение
		ОтменитьТранзакцию();
		ВызватьИсключение ОписаниеОшибки();
	КонецПопытки;

	Возврат НомерСписания + " (возврат " + НомерПеремещения + ")";
КонецФункции

// Документ «Списание товаров» по утилизированному возврату.
Функция СписатьВозврат(Баркод, ИмяСклада, Количество, ИдЗаказа, Площадка, ДатаДок, ХозОперация, Ответственный = "")
	// Пустая хоз. операция — ОТКАЗ, а не «спишем как-нибудь». Она решает, куда
	// лягут проводки, и документ без неё проведётся, а увидят это в отчётах
	// через месяц. ERROR приедет в ручной разбор сегодня.
	Если Не ЗначениеЗаполнено(ХозОперация) Тогда
		ВызватьИсключение "В задании не указана хозяйственная операция списания";
	КонецЕсли;

	СведенияОТоваре = НайтиНоменклатуруПоШтрихкоду(Баркод);
	Если СведенияОТоваре = Неопределено Тогда
		ВызватьИсключение "Не найдена номенклатура по штрихкоду " + Баркод;
	КонецЕсли;

	Склад = НайтиСклад(ИмяСклада);
	Документ = Документы.СписаниеТоваров.СоздатьДокумент();
	Если СтрДлина(СокрЛП(ДатаДок)) = 8 Тогда
		Документ.Дата = Дата(Число(Лев(ДатаДок, 4)), Число(Сред(ДатаДок, 5, 2)), Число(Прав(ДатаДок, 2)));
	Иначе
		Документ.Дата = ТекущаяДата();
	КонецЕсли;
	Документ.Склад = Склад;
	Документ.Магазин = Склад.Магазин;   // склад подчинён магазину, как у перемещения
	Документ.ХозяйственнаяОперация = ЗначениеСправочникаПоНаименованию(
		Метаданные.Документы.СписаниеТоваров.Реквизиты.ХозяйственнаяОперация.Тип,
		ХозОперация, "хоз. операция списания");
	// Комментарий начинается на "sync", и это не украшение: по нему
	// `ИсточникИзменения` метит движение как наше, а приложение такие дельты
	// отбрасывает. Здесь отбрасывать как раз НЕ надо — остаток ЦС от пары
	// документов не меняется, — но правило общее, и нарушать его в одном месте
	// значит завести исключение, о котором через полгода никто не вспомнит.
	Документ.Комментарий = "sync SCRAP " + ИдЗаказа + " " + Площадка;
	// Ответственный. Проверка «заполнен ли» тут не про пустое значение от
	// сайта — он всегда шлёт имя, умолчание у него в коде, — а про СТАРЫЙ
	// формат задания в девять полей: такое могло уехать до обновления и
	// доживать в очереди. Имя, которого нет в справочнике, — ОТКАЗ (внутри
	// `ЗначениеСправочникаПоНаименованию`): «Акт списания товаров» печатается с
	// ответственным, и тихо провести документ без него значит потерять след
	// навсегда, а ERROR приедет в ручной разбор сегодня.
	Если ЗначениеЗаполнено(Ответственный) Тогда
		Документ.Ответственный = ЗначениеСправочникаПоНаименованию(
			Метаданные.Документы.СписаниеТоваров.Реквизиты.Ответственный.Тип,
			Ответственный, "ответственный в документе списания");
	КонецЕсли;

	СтрокаТовара = Документ.Товары.Добавить();
	СтрокаТовара.Номенклатура = СведенияОТоваре.Номенклатура;
	СтрокаТовара.ХарактеристикаНоменклатуры = СведенияОТоваре.Характеристика;
	СтрокаТовара.Количество = Количество;
	СтрокаТовара.ЕдиницаИзмерения = СведенияОТоваре.Номенклатура.ЕдиницаХраненияОстатков;
	СтрокаТовара.Коэффициент = 1;
	СтрокаТовара.СтатьяЗатрат = ЗначениеСправочникаПоНаименованию(
		Метаданные.Документы.СписаниеТоваров.ТабличныеЧасти.Товары.Реквизиты.СтатьяЗатрат.Тип,
		"Списание товара", "статья затрат строки списания");
	// `КлючСтроки` связывает строку товара со строками табличной части
	// «Серийные номера». Мы их не пишем, связывать нечего, и строка у нас
	// всегда одна — то есть по смыслу ключ не нужен. Ставим его всё равно:
	// ноль в конфигурациях этого семейства означает «не задан», и окажись в
	// модуле проведения контроль на заполненность или на уникальность, документ
	// не провёлся бы, а узнали бы мы об этом на первой живой утилизации.
	// Номером строки он безвреден при любом раскладе и уникален, даже если строк
	// когда-нибудь станет больше одной.
	СтрокаТовара.КлючСтроки = Документ.Товары.Количество();
	Документ.Записать(РежимЗаписиДокумента.Проведение);

	Возврат Строка(Документ.Номер);
КонецФункции

// Элемент справочника по наименованию — БЕЗ жёстко прописанного имени справочника.
//
// Имя мы узнаём у МЕТАДАННЫХ самого реквизита, а не пишем в коде. Причина
// практическая: обработку правят в репозитории, где конфигурации нет, и всякое
// угаданное имя справочника («СтатьиЗатрат», «ХозяйственныеОперации») стало бы
// догадкой, которая либо не пройдёт синтаксический контроль, либо пройдёт и
// найдёт не то. Перебор справочников конфигурации стоит миллисекунды, а строк
// утилизации за день единицы.
Функция ЗначениеСправочникаПоНаименованию(ОписаниеТипа, Наименование, ЧтоИщем)
	Для Каждого МетаСправочник Из Метаданные.Справочники Цикл
		Если Не ОписаниеТипа.СодержитТип(Тип("СправочникСсылка." + МетаСправочник.Имя)) Тогда
			Продолжить;
		КонецЕсли;
		Ссылка = Справочники[МетаСправочник.Имя].НайтиПоНаименованию(Наименование, Истина);
		Если ЗначениеЗаполнено(Ссылка) Тогда
			Возврат Ссылка;
		КонецЕсли;
	КонецЦикла;
	// Отказ ВСЛУХ: документ с пустой хоз. операцией или пустой статьёй затрат
	// проведётся, а движение уйдёт не туда — и увидят это в отчётах через месяц.
	// Лучше вернуть ERROR, который приедет в ручной разбор сегодня.
	ВызватьИсключение "Не найдено значение """ + Наименование + """ (" + ЧтоИщем
		+ "). Заведите его в справочнике или поправьте название в обработке.";
КонецФункции

// Отмена неподтверждённого заказа — реверс проведённых перемещений на ЦС.
Функция ОтменитьПеремещенияЗаказа(ИдЗаказа)
	Запрос = Новый Запрос;
	// Реверсы ИСКЛЮЧАЕМ. Обратный документ получает комментарий
	// "sync REVERSE " + комментарий исходного, то есть содержит тот же order_id
	// и подходил под прежний шаблон: вторая отмена по тому же заказу отменила бы
	// и само перемещение, и его возврат — товар уехал бы на склад площадки
	// вместо возврата на ЦС.
	Запрос.Текст =
		"ВЫБРАТЬ Ссылка ИЗ Документ.ПеремещениеТоваров
		|ГДЕ Комментарий ПОДОБНО &Шаблон
		|	И НЕ Комментарий ПОДОБНО &ШаблонРеверса
		|	И Проведен = ИСТИНА";
	Запрос.УстановитьПараметр("Шаблон", "%order_id=" + ИдЗаказа + " %");
	Запрос.УстановитьПараметр("ШаблонРеверса", "sync REVERSE%");
	Выборка = Запрос.Выполнить().Выбрать();

	НомераОбратных = Новый Массив;
	Пока Выборка.Следующий() Цикл
		Исходный = Выборка.Ссылка.ПолучитьОбъект();
		СкладЦелевой = НайтиСклад(ПараметрыОбмена().СкладЦС);
		Обратный = Документы.ПеремещениеТоваров.СоздатьДокумент();
		Обратный.Дата = ТекущаяДата();
		Обратный.СкладОтправитель = Исходный.СкладПолучатель;
		Обратный.СкладПолучатель  = СкладЦелевой;
		Обратный.МагазинОтправитель = Исходный.МагазинПолучатель;
		Обратный.МагазинПолучатель  = СкладЦелевой.Магазин;
		Обратный.Комментарий = "sync REVERSE " + Исходный.Комментарий;
		Для Каждого СтрокаТовара Из Исходный.Товары Цикл
			НоваяСтрока = Обратный.Товары.Добавить();
			ЗаполнитьЗначенияСвойств(НоваяСтрока, СтрокаТовара);
		КонецЦикла;
		Обратный.Записать(РежимЗаписиДокумента.Проведение);
		НомераОбратных.Добавить(Строка(Обратный.Номер));
	КонецЦикла;

	Если НомераОбратных.Количество() = 0 Тогда
		Возврат "нет документов для отмены";
	КонецЕсли;
	Возврат СтрСоединить(НомераОбратных, ",");
КонецФункции

// Выгрузка остатков склада ЦС НА ЗАДАННОЕ ЧИСЛО (команда EXPORT_STOCK_ON_DATE).
// ДатаСтрокой — ГГГГММДД; остаток берётся на КОНЕЦ этого дня.
// Формат строки тот же, что у текущей выгрузки.
//
// Запрос намеренно СВОЙ, хотя от запроса ВыгрузитьОстаткиЦС отличается одним
// параметром: тот запрос — источник ТЕКУЩЕГО остатка, который уходит на площадки,
// база 1С только боевая, и проверить его правку негде. Общей у двух выгрузок
// сделана сборка строк — то есть форматы файлов разойтись не могут.
Функция ВыгрузитьОстаткиЦСНаДату(Пар, ДатаСтрокой)
	Если СтрДлина(СокрЛП(ДатаСтрокой)) <> 8 Тогда
		ВызватьИсключение "Неверная дата среза «" + ДатаСтрокой + "» — ожидается ГГГГММДД.";
	КонецЕсли;
	ДатаСреза = Дата(Число(Лев(ДатаСтрокой, 4)), Число(Сред(ДатаСтрокой, 5, 2)), Число(Прав(ДатаСтрокой, 2)));

	Запрос = Новый Запрос;
	Запрос.Текст =
		"ВЫБРАТЬ
		|	Остатки.Номенклатура КАК Номенклатура,
		|	Остатки.ХарактеристикаНоменклатуры КАК Характеристика,
		|	Остатки.КоличествоОстаток КАК Количество
		|ИЗ
		|	РегистрНакопления.ТоварыНаСкладах.Остатки(&Момент, Склад = &Склад) КАК Остатки";
	// Граница «Включая» на конец дня: без неё виртуальная таблица отдала бы остаток
	// на начало указанного момента, и движения самого выбранного числа (продажи,
	// приходы, перемещения) в срез не попали бы.
	Запрос.УстановитьПараметр("Момент", Новый Граница(КонецДня(ДатаСреза), ВидГраницы.Включая));
	Запрос.УстановитьПараметр("Склад", НайтиСклад(Пар.СкладЦС));
	Выборка = Запрос.Выполнить().Выбрать();

	Возврат СтрСоединить(СтрокиВыгрузкиОстатков(Выборка), Символы.ПС);
КонецФункции

// Сборка строк выгрузки остатков — общая для выгрузки «сейчас» и «на дату»,
// чтобы форматы двух файлов не разъехались.
// Строка: uid_1c|артикул|наименование|количество|баркод1,баркод2|размер|цвет
Функция СтрокиВыгрузкиОстатков(Выборка)
	Строки = Новый Массив;
	Пока Выборка.Следующий() Цикл
		Уид = ИдентификаторSKU(Выборка.Номенклатура, Выборка.Характеристика);
		Артикул = ОчиститьПоле(Выборка.Номенклатура.Артикул);
		// Наименование чистим ТОЛЬКО от переводов строки: на `|` внутри него
		// разбор рассчитан намеренно (режет хвост справа), а на разрыв строки —
		// нет, там товар просто исчезает из выгрузки.
		Наименование = СтрЗаменить(СтрЗаменить("" + Выборка.Номенклатура.Наименование,
			Символы.ПС, " "), Символы.ВК, " ");
		Кол = Формат(Выборка.Количество, "ЧГ=0; ЧДЦ=0");   // целое; отрицательное (пересортица) — с минусом
		Баркоды = БаркодыSKU(Выборка.Номенклатура, Выборка.Характеристика);
		Если ПустаяСтрока(Баркоды) Тогда
			Продолжить;
		КонецЕсли;
		Размер = ОчиститьПоле(Выборка.Характеристика);
		Цвет = "";
		Попытка
			Цвет = ОчиститьПоле(Выборка.Номенклатура.hiЦвет);
		Исключение
			Цвет = "";
		КонецПопытки;
		Строки.Добавить(Уид + "|" + Артикул + "|" + Наименование + "|" + Кол + "|" + Баркоды + "|" + Размер + "|" + Цвет);
	КонецЦикла;
	Возврат Строки;
КонецФункции

// Полный справочник баркодов (регистр «Штрихкоды»), независимо от остатка.
// Строка: uid_1c|артикул|наименование|баркод|размер|цвет
Функция ВыгрузитьСправочникБаркодов(Пар)
	Запрос = Новый Запрос;
	Запрос.Текст =
		"ВЫБРАТЬ
		|	Штрихкоды.Владелец КАК Номенклатура,
		|	Штрихкоды.ХарактеристикаНоменклатуры КАК Характеристика,
		|	Штрихкоды.Штрихкод КАК Штрихкод
		|ИЗ
		|	РегистрСведений.Штрихкоды КАК Штрихкоды
		|ГДЕ
		|	Штрихкоды.Штрихкод <> """"
		|	И Штрихкоды.Владелец ССЫЛКА Справочник.Номенклатура";
	Выборка = Запрос.Выполнить().Выбрать();

	Строки = Новый Массив;
	Пока Выборка.Следующий() Цикл
		Если ПустаяСтрока(Выборка.Штрихкод) Тогда
			Продолжить;
		КонецЕсли;
		Уид = ИдентификаторSKU(Выборка.Номенклатура, Выборка.Характеристика);
		Артикул = ОчиститьПоле(Выборка.Номенклатура.Артикул);

			Символы.ПС, " "), Символы.ВК, " ");
		Баркод = ОчиститьПоле(Выборка.Штрихкод);
		Размер = ОчиститьПоле(Выборка.Характеристика);
		Цвет   = ОчиститьПоле(Выборка.Номенклатура.hiЦвет);
		Строки.Добавить(Уид + "|" + Артикул + "|" + Наименование + "|" + Баркод + "|" + Размер + "|" + Цвет);
	КонецЦикла;

	Возврат СтрСоединить(Строки, Символы.ПС);

// Все баркоды размер-цвета через запятую.
// Текстовое поле для выгрузки: разделитель и переводы строк убираются.
//
// `|` внутри значения сдвигает поля при разборе: разбор режет строку СПРАВА по
// фиксированному числу полей, поэтому товар с остатком 4 и баркодом «4|4»
// читался как остаток 44 — десятикратное завышение уезжало на площадки как есть.
// Перевод строки хуже: он рвёт строку файла надвое, обе половины отбрасываются,
// товар ПРОПАДАЕТ из выгрузки — а часовая сверка идёт с missing_means_zero, и
// пропавшему товару проставляется ноль, то есть карточка обнуляется. Молча.
Функция ОчиститьПоле(Значение)
	Текст = СтрЗаменить("" + Значение, Символы.ПС, " ");
	Текст = СтрЗаменить(Текст, Символы.ВК, " ");
	Текст = СтрЗаменить(Текст, Символы.Таб, " ");
	Возврат СтрЗаменить(Текст, "|", " ");
КонецФункции

// ---- Оперативная отдача изменений остатка ЦС ----

// Публикует delta_ГГГГММДДЧЧММСС.txt с позициями, у которых остаток ЦС изменился
// с прошлой проверки. Нужна, чтобы правка в 1С доезжала до приложения за минуту,
// а не за час: полная выгрузка — это 152 тыс. позиций, чаще раза в час её не
// просят.
//
// Строка: баркод|новый остаток|источник|идентификатор
//
// Количество — АБСОЛЮТНЫЙ остаток на ЦС, а не приращение. Приращения копят
// ошибку навсегда, если файл потерялся или задвоился; абсолютное число
// самолечится следующим же сообщением.
//
// Источник "sync" — наши же документы (их комментарий начинается на "sync").
// Приложение такие строки отбрасывает: перемещение под заказ уже списало остаток
// в момент приёма заказа, применить его второй раз значит списать дважды.
//
// Идентификатор "<uuid документа>:<период движения>" — ключ от задвоения.
// Если прошлый прогон опубликовал файл, но не успел записать отметку времени,
// следующий выгрузит те же движения — приложение узнает их по ключу и не
// применит дважды. Оговорка: перепроведение документа той же датой даёт тот же
// ключ, и такое изменение приложение пропустит — его принесёт часовая выгрузка,
// то есть не позже чем через час.
//
// ДВЕ ГРАНИЦЫ, о которых надо знать честно:
//
// 1. Отбор идёт по Дв.Период, а это ДАТА ДОКУМЕНТА, а не момент его проведения.
//    Документ, проведённый СЕГОДНЯ задним числом, имеет Период в прошлом и в
//    окно не попадёт. На практике это бьёт почти только по нашим же
//    перемещениям (расчёт проводит их задним числом) — а они и так отбрасываются
//    как "sync". Обычные документы магазина проводятся текущей датой. Всё, что
//    сюда не попало, принесёт часовая выгрузка.
//
// 2. Частота дельты равна частоте запуска ВыполнитьОбмен. Если обработка
//    запускается раз в десять минут, то и остаток приезжает за десять минут, а
//    не за одну. Чтобы получить минуту, надо чаще запускать саму обработку.
Процедура ВыгрузитьИзменениеОстатковЦС(Пар)
	СкладЦС = НайтиСклад(Пар.СкладЦС);
	МоментС = ПрочитатьОтметкуДельты(Пар);
	МоментПо = ТекущаяДата();
	Если МоментС >= МоментПо Тогда
		Возврат;
	КонецЕсли;

	Запрос = Новый Запрос;
	Запрос.Текст =
		"ВЫБРАТЬ РАЗЛИЧНЫЕ
		|	Дв.Номенклатура КАК Номенклатура,
		|	Дв.ХарактеристикаНоменклатуры КАК Характеристика,
		|	Дв.Регистратор КАК Регистратор,
		|	Дв.Период КАК Период
		|ИЗ РегистрНакопления.ТоварыНаСкладах КАК Дв
		|ГДЕ Дв.Склад = &Склад И Дв.Период > &МоментС И Дв.Период <= &МоментПо";
	Запрос.УстановитьПараметр("Склад", СкладЦС);
	Запрос.УстановитьПараметр("МоментС", МоментС);
	Запрос.УстановитьПараметр("МоментПо", МоментПо);
	Результат = Запрос.Выполнить();

	// Предохранитель: на каждую изменившуюся позицию ниже идёт отдельный запрос
	// остатка, и на большом окне это тысячи запросов к боевой базе. Если позиций
	// слишком много (первый запуск, долгий простой обмена, массовая переоценка),
	// дельту пропускаем и двигаем отметку: этот остаток всё равно принесёт
	// часовая выгрузка — она одним запросом берёт весь склад и для больших
	// объёмов подходит лучше. Дельта существует ради СКОРОСТИ на единичных
	// изменениях, а не ради перекачки склада по одной позиции.
	Если Результат.Пустой() Тогда
		ЗаписатьОтметкуДельты(Пар, МоментПо);
		Возврат;
	КонецЕсли;
	ТаблицаИзменений = Результат.Выгрузить();
	Если ТаблицаИзменений.Количество() > МаксПозицийВДельте() Тогда
		ЗаписьЖурналаРегистрации("ОбменССайтом", УровеньЖурналаРегистрации.Информация, , ,
			"Дельта пропущена: изменилось " + ТаблицаИзменений.Количество()
			+ " позиций, это больше предела " + МаксПозицийВДельте()
			+ ". Остаток принесёт часовая выгрузка.");
		ЗаписатьОтметкуДельты(Пар, МоментПо);
		Возврат;
	КонецЕсли;

	Строки = Новый Массив;
	Для Каждого Выборка Из ТаблицаИзменений Цикл
		Баркоды = БаркодыSKU(Выборка.Номенклатура, Выборка.Характеристика);
		Если ПустаяСтрока(Баркоды) Тогда
			Продолжить;      // без баркода приложение позицию не опознает
		КонецЕсли;

		Источник = ИсточникИзменения(Выборка.Регистратор);
		Ключ = Строка(Выборка.Регистратор.УникальныйИдентификатор())
			+ ":" + Формат(Выборка.Период, "ДФ=ггггММддЧЧммсс");
		Остаток = ОстатокНаСкладе(СкладЦС, Выборка.Номенклатура, Выборка.Характеристика);
		Кол = Формат(Остаток, "ЧГ=0; ЧДЦ=0");

		Для Каждого Баркод Из СтрРазделить(Баркоды, ",") Цикл
			Если НЕ ПустаяСтрока(Баркод) Тогда
				Строки.Добавить(СокрЛП(Баркод) + "|" + Кол + "|" + Источник + "|" + Ключ);
			КонецЕсли;
		КонецЦикла;
	КонецЦикла;

	Если Строки.Количество() > 0 Тогда
		ОпубликоватьФайл(Пар.КаталогResults,
			"delta_" + Формат(МоментПо, "ДФ=ггггММддЧЧммсс") + ".txt",
			СтрСоединить(Строки, Символы.ПС));
	КонецЕсли;

	// Отметку двигаем ТОЛЬКО после успешной публикации: иначе сбой записи файла
	// потерял бы изменения безвозвратно. Лишний повтор безопаснее пропуска —
	// от повтора защищает ключ, от пропуска не защищает ничто.
	ЗаписатьОтметкуДельты(Пар, МоментПо);
КонецПроцедуры

// Сколько изменившихся позиций дельта берёт за один раз. Больше — пропускаем и
// ждём часовую выгрузку: она забирает весь склад одним запросом, а дельта на
// каждую позицию делает свой.
Функция МаксПозицийВДельте()
	Возврат 1000;

// Источник изменения: "sync" для наших же документов, иначе имя типа документа.
Функция ИсточникИзменения(Регистратор)
	Если ТипЗнч(Регистратор) = Тип("ДокументСсылка.ПеремещениеТоваров") Тогда
		Попытка
			Если СтрНачинаетсяС(СокрЛП("" + Регистратор.Комментарий), "sync") Тогда
				Возврат "sync";
			КонецЕсли;
		Исключение
			// комментарий недоступен — считаем документ чужим, это безопаснее:
			// чужой источник приложение применит, а "sync" молча отбросит
		КонецПопытки;
	КонецЕсли;
	Возврат СтрЗаменить("" + ТипЗнч(Регистратор), "|", " ");
КонецФункции

Функция ПрочитатьОтметкуДельты(Пар)
	Ф = Новый Файл(ПутьОтметкиДельты(Пар));
	Если НЕ Ф.Существует() Тогда
		// Первый запуск: берём час назад, а не начало времён — иначе первая же
		// дельта вышла бы размером с полную выгрузку.
		Возврат ТекущаяДата() - 3600;
	КонецЕсли;
	// Берём ТОЛЬКО ЦИФРЫ, а не строку целиком. ЗаписьТекста с КодировкаТекста.UTF8
	// пишет файл С BOM, и при чтении он остаётся в начале строки: проверка длины
	// тогда не сходится (15 символов вместо 14), отметка молча считается битой и
	// сбрасывается на «час назад» ПРИ КАЖДОМ запуске. Дельта от этого не врёт —
	// повторы отсекает ключ на стороне приложения, — но каждый прогон заново
	// перебирал бы часовое окно движений. Проверено на бою 19.09: в файле
	// действительно лежит BOM.
	Текст = ПрочитатьТекстФайла(ПутьОтметкиДельты(Пар));
	Цифры = "";
	Для Сч = 1 По СтрДлина(Текст) Цикл
		Симв = Сред(Текст, Сч, 1);
		Если Симв >= "0" И Симв <= "9" Тогда
			Цифры = Цифры + Симв;
		КонецЕсли;
	КонецЦикла;
	Если СтрДлина(Цифры) <> 14 Тогда
		Возврат ТекущаяДата() - 3600;
	КонецЕсли;
	Попытка
		Возврат Дата(Число(Лев(Цифры, 4)), Число(Сред(Цифры, 5, 2)), Число(Сред(Цифры, 7, 2)),
			Число(Сред(Цифры, 9, 2)), Число(Сред(Цифры, 11, 2)), Число(Прав(Цифры, 2)));
	Исключение
		Возврат ТекущаяДата() - 3600;
	КонецПопытки;
КонецФункции

// Атомарная публикация: пишем "имя.part", затем переименовываем в "имя".
